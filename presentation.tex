%--- Slide 1: What We Did ---
\begin{frame}{What We Did}
\textbf{The Input \& Data Sources}
\begin{itemize}
    \item \textbf{Raw CCTV Clips}: Analysed anonymised 1080p footage from 3 distinct angles (Entry/Exit, Main Floor, Billing Area).
    \item \textbf{Offline Data}: Correlated visual tracking data with offline Point-of-Sale (POS) transaction CSVs.
\end{itemize}

\vspace{0.3cm}
\textbf{Computer Vision Pipeline}
\begin{itemize}
    \item \textbf{Detection \& Tracking}: Implemented YOLOv8 for robust person detection and ByteTrack for consistent object tracking across frames.
    \item \textbf{Re-ID \& Zones}: Built deterministic visitor identification logic (color histograms) and mapped coordinates to physical store zones (Skincare, Fragrance, etc.).
    \item \textbf{Event Emission}: Generates highly structured, idempotent JSON events (\texttt{ENTRY}, \texttt{ZONE\_ENTER}, \texttt{BILLING\_QUEUE\_JOIN}).
\end{itemize}

\vspace{0.3cm}
\textbf{Intelligence API \& Dashboard}
\begin{itemize}
    \item \textbf{Backend}: Dockerised FastAPI application backed by SQLite to ingest events and compute live metrics.
    \item \textbf{Frontend}: Built a Live Web Dashboard streaming the real-time event feed and store KPIs.
\end{itemize}
\end{frame}

%--- Slide 2: Why We Did It ---
\begin{frame}{Why We Did It}
\textbf{Business Motivation}
\begin{itemize}
    \item While online channels have mature analytics (clicks, sessions, drop-offs), offline physical retail stores are historically a \textbf{data blind spot}.
    \item Automating camera analytics enables store managers to see real-time customer funnels, identical to an e-commerce website.
\end{itemize}

\vspace{0.3cm}
\textbf{Architectural Decisions}
\begin{itemize}
    \item \textbf{Idempotent Ingestion}: The \texttt{POST /events/ingest} endpoint guarantees idempotency via unique \texttt{event\_id}s, handling batch-processing failures gracefully.
    \item \textbf{Decoupled Pipeline}: Separated the heavy CV inference script from the lightweight API backend to ensure the API never blocks during video processing.
    \item \textbf{Graceful Degradation}: Returns structured HTTP 503 errors if the database goes offline, ensuring predictable behavior for client apps.
\end{itemize}

\vspace{0.3cm}
\textbf{Handling Edge Cases}
\begin{itemize}
    \item Explicitly detects and filters out \textbf{Store Staff} from customer metrics.
    \item Solved Re-entry inflation by merging rapid exits/re-entries.
\end{itemize}
\end{frame}

%--- Slide 3: Results ---
\begin{frame}{Results \& Deliverables}
\textbf{Key System Capabilities}
\begin{itemize}
    \item \textbf{Live Customer Funnel}: Tracks a physical session from \textit{Entry $\rightarrow$ Zone Visit $\rightarrow$ Billing Queue $\rightarrow$ Purchase}.
    \item \textbf{Automated POS Correlation}: Accurately identifies a "Purchase" by matching a visitor in the billing zone with a POS receipt timestamp within a 5-minute window.
    \item \textbf{Live Anomaly Detection}: Actively warns managers of "Dead Zones" (no visits in 30 mins) and "Queue Spikes".
\end{itemize}
